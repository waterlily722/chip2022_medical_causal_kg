import json
# py文件，上传原因更改后缀为txt文件
CASUAL_RELATION = 1
CONDITIONAL_RELATION = 2
HYPERNMY_RELATION = 3

# read train data
def read_json(file_name):
    with open(file_name, 'r') as input_data:
        block_list = json.load(input_data)
        for block in block_list:
            text, relation_of_mention_list = block['text'], block['relation_of_mention']
            for relation_of_mention in relation_of_mention_list:
                relation, head, tail = relation_of_mention['relation'], relation_of_mention['head'], relation_of_mention['tail']
                if relation in [CASUAL_RELATION, HYPERNMY_RELATION]:
                    head_entity, head_start_idx, head_end_idx = head['mention'], head['start_idx'], head['end_idx']
                    tail_entity, tail_start_idx, tail_end_idx = tail['mention'], tail['start_idx'], tail['end_idx']
                else: # CONDITIONAL_RELATION
                    head_entity, head_start_idx, head_end_idx = head['mention'], head['start_idx'], head['end_idx']
                    sub_relation, sub_head, sub_tail = tail['relation'], tail['head'], tail['tail']
                    assert sub_relation == CASUAL_RELATION
                    sub_head_entity, sub_head_start_idx, sub_head_end_idx = sub_head['mention'], sub_head['start_idx'], sub_head['end_idx']
                    sub_tail_entity, sub_tail_start_idx, sub_tail_end_idx = sub_tail['mention'], sub_tail['start_idx'], sub_tail['end_idx']

# write prediction result
def write_json(file_name):
    test_text_list = []
    with open('testA.json', 'r') as input_data:
        block_list = json.load(input_data)
        test_text_list = [block['text'] for block in block_list]

    output_result = []
    with open(file_name, 'w') as output_data:
        for text in test_text_list:
            block = {}
            block['text'] = text
            # implement your model predict code to replace dummy_predict
            block['relation_of_mention'] = dummy_predict(text)
            output_result.append(block)
        json.dump(output_result, output_data, indent=2, ensure_ascii=False)
